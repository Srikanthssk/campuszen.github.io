# Python meets biotechnology
from Bio.Seq import Seq

dna = Seq("ATGCGTACGTTAGC")
print(dna.reverse_complement())

# Analyze a DNA sequence
gc = (dna.count("G") +
      dna.count("C")) / len(dna)
print(f"GC: {gc:.1%}")
